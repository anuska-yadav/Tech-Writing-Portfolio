\[
\overbrace{a+b+c}^{\text{First Group}}+
\overbrace{x+y+z}^{\text{Second Group}}
=\overbrace{a+\underbrace{b+c}_{\text{Sum 1}}}^{\text{Total}}
\]

\section*{Piecewise Function}
\[
f(x)=\begin{cases}
x^2+2x+1,&x<0,\\
\sqrt{x+1},&0\leq x<4,\\
x-1,&x\geq4.
\end{cases}
\]
